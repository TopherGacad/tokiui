## Indeterminate

Set `checked="indeterminate"` to show the dash state — useful for parent checkboxes in a tree where only some children are selected.

<Showcase title="Indeterminate" code={`'use client'
import { useState } from 'react'

const PERMISSIONS = ['Read files', 'Write files', 'Delete files']

export function FilePermissions() {
  const [checked, setChecked] = useState([true, false, false])

  const allChecked = checked.every(Boolean)
  const someChecked = checked.some(Boolean)
  const parentState = allChecked ? true : someChecked ? 'indeterminate' : false

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
      <div style={{ display: 'flex', gap: 10, alignItems: 'center' }}>
        <Checkbox
          id="all-perms"
          checked={parentState}
          onCheckedChange={() => setChecked(checked.fill(!allChecked).slice())}
        />
        <label htmlFor="all-perms" style={{ fontSize: 14, fontWeight: 600 }}>
          File permissions
        </label>
      </div>
      <div style={{ paddingLeft: 26, display: 'flex', flexDirection: 'column', gap: 8 }}>
        {PERMISSIONS.map((perm, i) => (
          <div key={perm} style={{ display: 'flex', gap: 10, alignItems: 'center' }}>
            <Checkbox
              id={\`perm-\${i}\`}
              checked={checked[i]}
              onCheckedChange={(val) =>
                setChecked(checked.map((c, idx) => (idx === i ? !!val : c)))
              }
            />
            <label htmlFor={\`perm-\${i}\`} style={{ fontSize: 14 }}>{perm}</label>
          </div>
        ))}
      </div>
    </div>
  )
}`}>
  <CheckboxIndeterminatePreview />
</Showcase>

## Checkbox group (card style)

Combine multiple checkboxes in a card-style list. Highlight the selected card with a primary border and tinted background.

<Showcase title="Checkbox group" code={`'use client'
import { useState } from 'react'

const FRAMEWORKS = [
  { id: 'next',   label: 'Next.js',   desc: 'React framework'  },
  { id: 'nuxt',   label: 'Nuxt',      desc: 'Vue framework'    },
  { id: 'astro',  label: 'Astro',     desc: 'Content sites'    },
  { id: 'remix',  label: 'Remix',     desc: 'Full-stack React' },
  { id: 'svelte', label: 'SvelteKit', desc: 'Svelte framework' },
]

export function FrameworkPicker() {
  const [selected, setSelected] = useState(['next'])

  function toggle(id) {
    setSelected((prev) =>
      prev.includes(id) ? prev.filter((s) => s !== id) : [...prev, id]
    )
  }

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 4, maxWidth: 320 }}>
      {FRAMEWORKS.map((fw) => (
        <label
          key={fw.id}
          style={{
            display: 'flex', alignItems: 'center', gap: 10,
            padding: '10px 12px', borderRadius: 'var(--radius-sm)',
            border: \`1px solid \${selected.includes(fw.id) ? 'var(--primary)' : 'var(--border)'}\`,
            background: selected.includes(fw.id)
              ? 'color-mix(in oklch, var(--primary) 6%, var(--background))'
              : 'var(--background)',
            cursor: 'pointer', transition: 'all 0.15s',
          }}
        >
          <Checkbox
            checked={selected.includes(fw.id)}
            onCheckedChange={() => toggle(fw.id)}
          />
          <span>
            <span style={{ display: 'block', fontSize: 14, fontWeight: 500 }}>{fw.label}</span>
            <span style={{ display: 'block', fontSize: 12, color: 'var(--muted-foreground)' }}>{fw.desc}</span>
          </span>
        </label>
      ))}
    </div>
  )
}`}>
  <CheckboxGroupPreview />
</Showcase>

## API

### Checkbox

| Prop | Type | Default | Description |
|---|---|---|---|
| `size` | `sm \| default \| lg` | `default` | Checkbox size |
| `color` | `default \| success \| warning \| info \| error` | `default` | Checked/indeterminate color |
| `error` | `boolean` | `false` | Applies destructive border |
| `checked` | `boolean \| 'indeterminate'` | — | Controlled checked state |
| `defaultChecked` | `boolean` | `false` | Uncontrolled initial state |
| `onCheckedChange` | `(checked: boolean \| 'indeterminate') => void` | — | Called when state changes |
| `disabled` | `boolean` | `false` | Prevents interaction |
| `required` | `boolean` | `false` | Marks the input as required in a form |
| `name` | `string` | — | Input name for form submission |
| `value` | `string` | `'on'` | Value submitted in a form when checked |
| `className` | `string` | — | Additional class names |

### CheckboxGroup

| Prop | Type | Default | Description |
|---|---|---|---|
| `size` | `sm \| default \| lg` | `default` | Inherited by all child Checkboxes |
| `color` | `default \| success \| warning \| info \| error` | `default` | Inherited by all child Checkboxes |
| `error` | `boolean` | `false` | Inherited by all child Checkboxes |
| `orientation` | `vertical \| horizontal` | `vertical` | Layout direction |
| `className` | `string` | — | Additional class names |

## Accessibility

- Rendered as `<button role="checkbox">` — keyboard activatable with `Space`
- `aria-checked` reflects the current state including `"mixed"` for indeterminate
- Connect a visible `<label>` using `htmlFor` matching the checkbox `id`
- Disabled checkboxes use `aria-disabled="true"` and are excluded from tab order
