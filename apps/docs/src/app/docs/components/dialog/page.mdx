## Sizes

The `size` prop on `DialogContent` sets the max width. All sizes are full-width on small screens.

| Size | Max width | Use for |
|---|---|---|
| `sm` | 384 px | Confirmations, short messages |
| `default` | 512 px | Forms, standard content |
| `lg` | 672 px | Data-heavy content, multi-column forms |
| `full` | viewport − 48 px | Focused workflows that need maximum space |

## Loading state

Control the open state manually so you can block dismiss while an async action is in flight. Use `aria-busy` on the submit button and disable both actions during the request.

<Showcase title="Loading state" code={`'use client'
import { useState } from 'react'

export function SaveDialog() {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleSave() {
    setLoading(true)
    await saveToCloud()
    setLoading(false)
    setOpen(false)
  }

  return (
    <Dialog open={open} onOpenChange={(o) => { if (!loading) setOpen(o) }}>
      <DialogTrigger asChild>
        <Button onClick={() => setOpen(true)}>Save to cloud</Button>
      </DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Save changes</DialogTitle>
          <DialogDescription>
            Your work will be saved to the cloud and synced across devices.
          </DialogDescription>
        </DialogHeader>
        <DialogFooter>
          <Button variant="outline" color="neutral" disabled={loading} onClick={() => setOpen(false)}>
            Cancel
          </Button>
          <Button disabled={loading} aria-busy={loading} onClick={handleSave}>
            {loading ? 'Saving…' : 'Save'}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}`}>
  <LoadingDialogPreview />
</Showcase>

## Unsaved changes guard

Prevent accidental dismissal when a form is dirty. Block `onInteractOutside` and `onEscapeKeyDown` until the user explicitly saves or discards.

<Showcase title="Unsaved changes guard" code={`'use client'
import { useState } from 'react'

export function GuardedDialog() {
  const [open, setOpen] = useState(false)
  const [dirty, setDirty] = useState(false)

  function handleOpenChange(next: boolean) {
    if (!next && dirty) return  // block close while dirty
    setOpen(next)
    if (!next) setDirty(false)
  }

  return (
    <Dialog open={open} onOpenChange={handleOpenChange}>
      <DialogTrigger asChild>
        <Button variant="outline" onClick={() => setOpen(true)}>Edit with guard</Button>
      </DialogTrigger>
      <DialogContent
        onInteractOutside={(e) => e.preventDefault()}
        onEscapeKeyDown={(e) => { if (dirty) e.preventDefault() }}
      >
        <DialogHeader>
          <DialogTitle>Unsaved changes guard</DialogTitle>
          <DialogDescription>
            Backdrop clicks and Escape are blocked while you have unsaved changes.
          </DialogDescription>
        </DialogHeader>
        <Input
          onChange={(e) => setDirty(e.target.value.length > 0)}
          placeholder="Type to mark as dirty…"
        />
        <DialogFooter>
          <Button variant="outline" color="neutral" onClick={() => { setDirty(false); setOpen(false) }}>
            Discard
          </Button>
          <Button onClick={() => { setDirty(false); setOpen(false) }}>Save</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}`}>
  <NoBackdropDialogPreview />
</Showcase>

## Multi-step wizard

Manage a `step` state inside the dialog. Reset it on close with a short `setTimeout` so the animation completes before the step resets.

<Showcase title="Multi-step wizard" code={`'use client'
import { useState } from 'react'

const STEPS = ['Account', 'Profile', 'Review']

export function WizardDialog() {
  const [step, setStep] = useState(1)
  const total = STEPS.length

  function reset() { setTimeout(() => setStep(1), 200) }

  return (
    <Dialog onOpenChange={(open) => { if (!open) reset() }}>
      <DialogTrigger asChild>
        <Button>Create account</Button>
      </DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <div className="flex justify-between items-center">
            <DialogTitle>{STEPS[step - 1]}</DialogTitle>
            <span className="text-xs text-muted-foreground font-mono">{step} / {total}</span>
          </div>
          <DialogDescription>
            {step === 1 && 'Set up your login credentials.'}
            {step === 2 && 'Tell us how to address you.'}
            {step === 3 && 'Review your details before finishing.'}
          </DialogDescription>
        </DialogHeader>
        {/* Progress bar */}
        <div className="h-[3px] bg-muted rounded-full overflow-hidden">
          <div
            className="h-full bg-primary rounded-full transition-[width] duration-300"
            style={{ width: \`\${(step / total) * 100}%\` }}
          />
        </div>
        {/* Step content */}
        {step === 1 && <div className="grid gap-3">{/* email + password inputs */}</div>}
        {step === 2 && <div className="grid gap-3">{/* name + username inputs */}</div>}
        {step === 3 && <div className="bg-muted rounded-lg p-4 text-sm text-muted-foreground">Review summary…</div>}
        <DialogFooter>
          {step > 1 ? (
            <Button variant="outline" color="neutral" onClick={() => setStep((s) => s - 1)}>Back</Button>
          ) : (
            <DialogClose asChild>
              <Button variant="outline" color="neutral">Cancel</Button>
            </DialogClose>
          )}
          {step < total ? (
            <Button onClick={() => setStep((s) => s + 1)}>Next</Button>
          ) : (
            <DialogClose asChild><Button>Finish</Button></DialogClose>
          )}
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}`}>
  <WizardDialogPreview />
</Showcase>

## API

### DialogContent

| Prop | Type | Default | Description |
|---|---|---|---|
| `size` | `'sm' \| 'default' \| 'lg' \| 'full'` | `'default'` | Max width of the dialog panel |
| `scrollBehavior` | `'outside' \| 'inside'` | `'outside'` | `outside` scrolls the whole dialog; `inside` fixes header/footer and scrolls only `DialogBody` |
| `fullscreenMobile` | `boolean` | `false` | On small screens, slides up from the bottom edge as a sheet instead of centering |
| `className` | `string` | — | Merged with the panel's class list |
| All Radix `Content` props | | | `onOpenAutoFocus`, `onCloseAutoFocus`, `onInteractOutside`, `onEscapeKeyDown`, … |

### Dialog (root)

| Prop | Type | Default | Description |
|---|---|---|---|
| `open` | `boolean` | — | Controlled open state |
| `onOpenChange` | `(open: boolean) => void` | — | Called when open state changes |
| `defaultOpen` | `boolean` | `false` | Uncontrolled initial state |
| `modal` | `boolean` | `true` | Whether to use modal behavior (focus trap + scroll lock) |

## Sub-components

| Component | Description |
|---|---|
| `Dialog` | Root state container |
| `DialogTrigger` | Element that opens the dialog — use `asChild` to avoid a wrapper |
| `DialogContent` | Panel rendered inside the overlay — includes the built-in close button |
| `DialogOverlay` | The backdrop — rendered automatically by `DialogContent` |
| `DialogHeader` | Top section wrapping title and description |
| `DialogTitle` | Required accessible title (becomes `aria-labelledby`) |
| `DialogDescription` | Optional accessible description (becomes `aria-describedby`) |
| `DialogBody` | Scrollable middle section — use with `overflow-hidden` on `DialogContent` |
| `DialogFooter` | Bottom row for actions — reverses column order on mobile |
| `DialogClose` | Element that closes the dialog — use `asChild` to wrap your own button |
| `DialogPortal` | Renders content into `document.body` — used internally by `DialogContent` |

## Accessibility

- Focus is trapped inside the dialog while open
- Pressing `Escape` closes the dialog (block with `onEscapeKeyDown={(e) => e.preventDefault()}`)
- Scroll is locked on the body while open
- `DialogTitle` is required — screen readers announce it when the dialog opens
- `DialogDescription` is optional but strongly recommended for context
- Use `DialogTitle` with `className="sr-only"` when no visible title is needed (action sheets, share menus)
- All interactive elements inside the dialog are reachable with `Tab`
